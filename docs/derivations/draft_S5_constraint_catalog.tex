\paragraph{Representative constraint templates.}
Each pattern compiles into a guarded constraint over one trace relation; we give five representatives (full 16 in Table~\ref{tab:constraint-catalog}).

\begin{description}
\item[T-P3 Cross-rank replication.] \textit{Relation} \texttt{tensor\_state}; \textit{cols} (dp, step, name, cksum, shard\_flag); \textit{$G_{\text{topo}}$} non-sharded AND same parallel group; \textit{$G_{\text{phase}}$} \texttt{stage='after-optimizer-step'}; \textit{V} \texttt{COUNT(DISTINCT cksum) > 1} per (name, step). Coverage 42 (10.7\%).
\item[T-P4 Invocation frequency.] \textit{Relation} \texttt{call\_event}; \textit{cols} (hookpoint, step, count); \textit{$G_{\text{phase}}$} per training step; \textit{V} \texttt{count <> K}. Coverage 19 (4.8\%).
\item[T-P9 Init distribution.] \textit{Relation} \texttt{tensor\_state}; \textit{cols} (name, std/var, fan\_in); \textit{$G_{\text{phase}}$} \texttt{stage='init'}; \textit{V} \texttt{std} deviates from declared (e.g.\ $\sqrt{d_{\text{model}}}$ scaling). Coverage 19 (4.8\%).
\item[T-P10 Config-implied coupling.] \textit{Relation} \texttt{optimizer\_state} $\bowtie$ config; \textit{cols} (config\_flag, actual\_state); \textit{$G$} static; \textit{V} flag inconsistent with state (e.g.\ \texttt{zero\_stage>0} $\Rightarrow$ sharded optimizer state). Coverage 45 (11.5\%).
\item[T-P16 Loss normalization.] \textit{Relation} \texttt{loss\_event}; \textit{cols} (divisor, reduction\_mode, numel); \textit{$G_{\text{phase}}$} at loss reduction; \textit{V} \texttt{divisor <> reduction\_semantics}. Coverage 11 (2.8\%).
\end{description}
